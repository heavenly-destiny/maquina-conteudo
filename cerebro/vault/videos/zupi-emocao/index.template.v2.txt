      const T = __TIMING__;
      const W = __WORDS__;
      const tl = gsap.timeline({ paused: true });
      const w = (ln, i) => W[ln][i][1];

      // ---- motion energy follows the emotion map: same language, different speed ----
      const ENERGY = {
        melancholy: { d: 0.9, ease: "sine.out", blur: 10, y: 14, rx: 0 },
        suspense: { d: 0.7, ease: "power2.out", blur: 14, y: 0, rx: 0, scale: 1.12 },
        hope: { d: 0.55, ease: "power3.out", blur: 14, y: 26, rx: 40 },
        triumph: { d: 0.42, ease: "expo.out", blur: 16, y: 40, rx: 70 },
        relief: { d: 0.5, ease: "power3.out", blur: 14, y: 24, rx: 45 },
        authority: { d: 0.6, ease: "expo.out", blur: 14, y: 60, rx: 70 },
      };
      const wordIn = (sel, t, mood) => {
        const e = ENERGY[mood];
        tl.fromTo(sel, { opacity: 0, y: e.y, rotationX: e.rx, scale: e.scale || 1, filter: `blur(${e.blur}px)`, transformOrigin: "50% 100%" },
          { opacity: 1, y: 0, rotationX: 0, scale: 1, filter: "blur(0px)", duration: e.d, ease: e.ease }, t - 0.07);
      };
      const out = (sel, t, d = 0.3) => tl.to(sel, { opacity: 0, filter: "blur(12px)", duration: d, ease: "power2.in" }, t);
      const camera = (sel, t0, t1, s, ease = "none") => tl.fromTo(sel, { scale: 1 }, { scale: s, duration: t1 - t0, ease }, t0);
      const float3d = (sel, t0, t1, amp, rot, blur) =>
        tl.fromTo(sel, { y: amp, rotation: -rot, rotationY: -12, filter: `blur(${blur}px)` }, { y: -amp, rotation: rot, rotationY: 12, filter: `blur(${blur}px)`, duration: t1 - t0, ease: "sine.inOut" }, t0);
      const sweep = (sel, t, d = 0.8) => tl.fromTo(sel, { xPercent: -160 }, { xPercent: 420, duration: d, ease: "power2.inOut" }, t);
      const money = (el, from, to, t, d, ease, prefix = "R$ ") => {
        const o = { v: from }, node = document.querySelector(el);
        const f = () => (node.textContent = prefix + o.v.toLocaleString("pt-BR", { minimumFractionDigits: o.dec ? 2 : 2, maximumFractionDigits: 2 }));
        tl.fromTo(o, { v: from }, { v: to, duration: d, ease, onUpdate: f, immediateRender: false }, t);
      };
      const hblur = (id, from, to, t, d, ease) => {
        const g = document.getElementById(id), p = { v: from };
        const set = () => g.setAttribute("stdDeviation", p.v + " 0");
        set();
        tl.fromTo(p, { v: from }, { v: to, duration: d, ease, onUpdate: set }, t);
      };
      const dust = (id, t0, t1, dir = -1, n = 9) => {
        const box = document.getElementById(id);
        for (let i = 0; i < n; i++) {
          const d = document.createElement("div");
          d.className = "dust";
          const s = 14 + ((i * 37) % 40);
          Object.assign(d.style, { width: s + "px", height: s + "px", left: ((i * 211) % 1800) + 60 + "px", top: ((i * 131) % 900) + 80 + "px" });

          tl.fromTo(d, { y: 0, opacity: 0.2 + (i % 3) * 0.15 }, { y: dir * (60 + (i % 4) * 30), x: (i % 2 ? 1 : -1) * 30, opacity: 0.08, duration: t1 - t0, ease: "sine.inOut" }, t0);
        }
      };

      dust("s1-dust", 0, T.xfade_suspense + 0.8, 1, 6);
      camera("#s1-in", 0, T.xfade_suspense + 0.8, 1.04, "sine.inOut");

      ["#dr1", "#dr2", "#dr3"].forEach((s, i) => {

      // the hit: one word slams, everything else holds still
      tl.fromTo("#b6", { opacity: 0, scale: 1.6, filter: "blur(20px)" }, { opacity: 1, scale: 1, filter: "blur(0px)", duration: 0.32, ease: "power4.out" }, T.quebrou - 0.05);
      tl.fromTo("#s2-in", { x: 0 }, { x: 8, duration: 0.05, yoyo: true, repeat: 3, ease: "none" }, T.quebrou);
      tl.fromTo("#alarm", { opacity: 0, y: 60, rotationX: -30 }, { opacity: 1, y: 0, rotationX: 0, duration: 0.6, ease: "power3.out" }, T.quebrou + 0.25);
      out("#s2-headB", w(4, 0) - 0.15);
      tl.to("#alarm", { y: 100, opacity: 0.0, filter: "blur(8px)", duration: 0.6, ease: "power2.in" }, w(4, 0) - 0.15);
      // "nada guardado": the balance pulses like a heartbeat
      tl.fromTo("#zero", { opacity: 0, scale: 0.9, filter: "blur(14px)" }, { opacity: 1, scale: 1, filter: "blur(0px)", duration: 0.7, ease: "sine.out" }, w(4, 0));

      tl.fromTo("#s3-in", { opacity: 0, filter: "brightness(0.6) saturate(0.2)" }, { opacity: 1, filter: "brightness(1) saturate(1)", duration: 1.0, ease: "sine.out" }, T.hope_in);
      tl.fromTo("#s3-glow", { scale: 0.4, opacity: 0 }, { scale: 1.2, opacity: 1, duration: 3.2, ease: "sine.out" }, T.hope_in);
      dust("s3-dust", T.hope_in, T.push_triumph + 0.3, -1, 10);
      camera("#s3-in", T.hope_in, T.push_triumph + 0.3, 1.04);

      wordIn("#c2", w(5, 5), "hope");
      tl.fromTo("#box", { opacity: 0, y: 60, rotationX: 24, filter: "blur(12px)" }, { opacity: 1, y: 0, rotationX: 0, filter: "blur(0px)", duration: 0.8, ease: "power3.out" }, T.hope_in + 0.5);
      sweep("#g-box", T.hope_in + 1.2);
